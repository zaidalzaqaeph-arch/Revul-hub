local function applyProps(inst, props)
	for k, v in pairs(props) do
		if k ~= "Parent" then inst[k] = v end
	end
	if props.Parent then inst.Parent = props.Parent end
	return inst
end

local function new(class, props)
	return applyProps(Instance.new(class), props)
end

local function tween(obj, t, props, style)
	TweenService:Create(obj, TweenInfo.new(t, style or Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props):Play()
end

local function fileWrite(name, data)
	local ok = pcall(function() writefile(name, data) end)
	if ok then return true end
	ok = pcall(function() if syn and syn.write_file then syn.write_file(name, data) end end)
	return ok
end
local function fileRead(name)
	local s
	pcall(function() s = readfile(name) end)
	if type(s) ~= "string" then
		pcall(function() if syn and syn.read_file then s = syn.read_file(name) end end)
	end
	return s
end
local function saveUDim(name, pos)
	if not name or not pos then return end
	fileWrite(name, string.format("%.4f,%.1f,%.4f,%.1f", pos.X.Scale, pos.X.Offset, pos.Y.Scale, pos.Y.Offset))
end
local function loadUDim(name, fallback)
	local s = fileRead(name)
	if type(s) == "string" and s ~= "" then
		local parts = {}
		for part in string.gmatch(s, "[^,]+") do table.insert(parts, part) end
		if #parts >= 4 then
			local xs, xo, ys, yo = tonumber(parts[1]), tonumber(parts[2]), tonumber(parts[3]), tonumber(parts[4])
			if xs and xo and ys and yo then return UDim2.new(xs, xo, ys, yo) end
		end
	end
	return fallback
end

local function isRagdoll(hum)
	if not hum then return false end
	local st = hum:GetState()
	return st == Enum.HumanoidStateType.Physics
		or st == Enum.HumanoidStateType.Ragdoll
		or st == Enum.HumanoidStateType.FallingDown
		or hum.PlatformStand == true
end

local function findBat()
	local char = LP.Character
	if not char then return nil end
	local function scan(parent)
		for _, name in ipairs(BAT_NAMES) do
			local t = parent:FindFirstChild(name)
			if t and t:IsA("Tool") then return t end
		end
		for _, t in ipairs(parent:GetChildren()) do
			if t:IsA("Tool") then
				local n = t.Name:lower()
				if n:find("bat", 1, true) or n:find("slap", 1, true) then return t end
			end
		end
		return nil
	end
	return scan(char) or scan(LP:FindFirstChild("Backpack") or char)
end

local function swingBat(bat)
	if not bat then return end
	pcall(function() bat:Activate() end)
	local remote = bat:FindFirstChildOfClass("RemoteEvent") or bat:FindFirstChildWhichIsA("RemoteEvent", true)
	if remote then pcall(function() remote:FireServer() end) end
end

local function isCombatTool(name)
	local n = tostring(name or ""):lower()
	if n:find("bat", 1, true) or n:find("slap", 1, true) or n:find("medusa", 1, true) or n:find("glove", 1, true) then return true end
	for _, bn in ipairs(BAT_NAMES) do
		if name == bn then return true end
	end
	return false
end

local function holdingBrainrot()
	local char = LP.Character
	if not char then return false end
	local function attrOn(inst, key)
		local ok, value = pcall(function() return inst:GetAttribute(key) end)
		return ok and value == true
	end
	if attrOn(LP, "Stealing") or attrOn(LP, "AntiKick") or attrOn(char, "Stealing") or attrOn(char, "AntiKick") then
		return true
	end
	local okC, carrying = pcall(function() return LP:GetAttribute("Carrying") or char:GetAttribute("Carrying") end)
	if okC and carrying then return true end
	for _, child in ipairs(char:GetChildren()) do
		if child:IsA("Tool") then
			if not isCombatTool(child.Name) then return true end
			local name = child.Name:lower()
			if name:find("brainrot", 1, true) or name:find("skibidi", 1, true) or name:find("toilet", 1, true)
				or name:find("animal", 1, true) or name:find("carry", 1, true) or name:find("grab", 1, true) then
				return true
			end
		elseif child:IsA("Model") then
			local name = child.Name:lower()
			if name:find("brainrot", 1, true) or name:find("skibidi", 1, true) or name:find("toilet", 1, true)
				or name:find("animal", 1, true) or name:find("carry", 1, true) or name:find("grab", 1, true) then
				return true
			end
		end
	end
	return false
end

	local function stripOther(char)
		if not char then return end
		for _, part in ipairs(char:GetDescendants()) do
			if part:IsA("BasePart") then pcall(function() part.CanCollide = false end) end
		end
	end
	pcall(function()
		for _, plr in ipairs(Players:GetPlayers()) do
			if plr ~= LP and plr.Character then stripOther(plr.Character) end
		end
		Players.PlayerAdded:Connect(function(plr)
			plr.CharacterAdded:Connect(function(char) task.wait(0.15) stripOther(char) end)
			if plr.Character then stripOther(plr.Character) end
		end)
	end)
	local function capHoriz(root, maxH)
		pcall(function()
			local v = root.AssemblyLinearVelocity
			if v.X ~= v.X or v.Y ~= v.Y or v.Z ~= v.Z then
				root.AssemblyLinearVelocity = Vector3.zero
				return
			end
			local horiz = Vector3.new(v.X, 0, v.Z)
			local mag = horiz.Magnitude
			if mag > maxH and mag > 0 then
				local scale = maxH / mag
				root.AssemblyLinearVelocity = Vector3.new(v.X * scale, v.Y, v.Z * scale)
			end
			local av = root.AssemblyAngularVelocity
			if av.Magnitude > 45 or av.X ~= av.X then
				root.AssemblyAngularVelocity = Vector3.zero
			end
		end)
	end
	RunService.Heartbeat:Connect(function()
		if State.autoBat then return end
		local root = getHRP()
		if not root then return end
		local maxH = (State.autoLeft or State.autoRight) and 80 or 160
		capHoriz(root, maxH)
	end)
end

local DEFAULT_WALK = 16
local function restoreWalkSpeed()
	local hum = getHum()
	if not hum then return end
	pcall(function()
		if math.abs((hum.WalkSpeed or 16) - DEFAULT_WALK) > 0.05 then
			hum.WalkSpeed = DEFAULT_WALK
		end
	end)
end

	local lastMoveDir = Vector3.new()
	local function flatMove(hum)
		local md = hum.MoveDirection
		if md.Magnitude > 0.05 then
			local f = Vector3.new(md.X, 0, md.Z)
			if f.Magnitude > 0 then return f.Unit end
		end
		local cam = workspace.CurrentCamera
		if not cam then return nil end
		local v = Vector3.zero
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then v += Vector3.new(cam.CFrame.LookVector.X, 0, cam.CFrame.LookVector.Z) end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then v -= Vector3.new(cam.CFrame.LookVector.X, 0, cam.CFrame.LookVector.Z) end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then v -= Vector3.new(cam.CFrame.RightVector.X, 0, cam.CFrame.RightVector.Z) end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then v += Vector3.new(cam.CFrame.RightVector.X, 0, cam.CFrame.RightVector.Z) end
		if v.Magnitude > 0.05 then return v.Unit end
		return nil
	end
	RunService.RenderStepped:Connect(function(dt)
		if State.autoLeft or State.autoRight or State.autoBat then return end
		local hum, root = getHum(), getHRP()
		if not hum or not root or hum.Health <= 0 or isRagdoll(hum) then return end
		local carrying = holdingBrainrot()
		local autoCarry = (State.autoCarrySwitch == true) and carrying and not State.lagger
		local useCarry = (State.carryMode or autoCarry) and not State.lagger
		if not State.speedEnabled and not useCarry and not State.lagger then
			return
		end
		local dir = flatMove(hum)
		if not dir then return end
		lastMoveDir = dir
		local spd
		if State.lagger then
			spd = tonumber(State.laggerSpeed) or 15
		elseif useCarry then
			spd = tonumber(State.carrySpeed) or 30
		else
			spd = tonumber(State.customSpeed) or 29
		end
		if spd ~= spd then spd = DEFAULT_WALK end
		spd = math.clamp(spd, 1, 500)
		pcall(function()
			if useCarry then
				local walk = tonumber(hum.WalkSpeed) or 0
				if walk ~= walk or walk < 0 then walk = 0 end
				local extra = spd - walk
				if extra < 0 then extra = 0 end
				local d = math.clamp(dt or 0, 0, 1 / 20)
				if extra > 0 then
					root.CFrame = root.CFrame + Vector3.new(lastMoveDir.X * extra * d, 0, lastMoveDir.Z * extra * d)
				end
			else
				if not dir then return end
				if hum.WalkSpeed ~= spd then hum.WalkSpeed = spd end
				local v = root.AssemblyLinearVelocity
				root.AssemblyLinearVelocity = Vector3.new(lastMoveDir.X * spd, v.Y, lastMoveDir.Z * spd)
			end
		end)
	end)
end

		if not State.infJump then return end

		if not root then return end

			or UserInputService:IsKeyDown(Enum.KeyCode.ButtonA)

				root.Velocity = Vector3.new(root.Velocity.X, 38, root.Velocity.Z)
			end
		end
	end)

local function startFlight()
	clearConn("Flight")
	local bodyVel
	Connections.Flight = RunService.Heartbeat:Connect(function()
		local hrp = getHRP()
		if not hrp then return end
		local cam = workspace.CurrentCamera
		if not cam then return end
		if not bodyVel or bodyVel.Parent ~= hrp then
			if bodyVel then pcall(function() bodyVel:Destroy() end) end
			bodyVel = Instance.new("BodyVelocity")
			bodyVel.MaxForce = Vector3.new(1e5, 1e5, 1e5)
			bodyVel.Parent = hrp
		end
		local move = Vector3.zero
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then move += cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then move -= cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then move -= cam.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then move += cam.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0, 1, 0) end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then move -= Vector3.new(0, 1, 0) end
		local spd = State.speedEnabled and State.customSpeed or 50
		bodyVel.Velocity = (move.Magnitude > 0) and (move.Unit * spd) or Vector3.zero
	end)
end

local function stopFlight()

	local hrp = getHRP()

		for _, v in ipairs(hrp:GetChildren()) do
			if v:IsA("BodyVelocity") then v:Destroy() end
		end
	end
end

					root.Anchored = false

					if not State.autoBat then root.AssemblyLinearVelocity = Vector3.zero end
				end
			end)
		end

local TP_FLOOR = -7.00
local lastAutoTpAt = 0

local function tpDownNow(force)
	local char = LP.Character
	if not char then return end
	local hrp = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso")
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hrp or not hum then return end
	local carrying = holdingBrainrot()
	if not force then
		if tick() - lastAutoTpAt < 0.12 then return end
		if not carrying then return end
		local st = hum:GetState()
		local inAir = hum.FloorMaterial == Enum.Material.Air
			or st == Enum.HumanoidStateType.Jumping
			or st == Enum.HumanoidStateType.Freefall
		if not inAir then return end
		local rad = tonumber(State.autoTpDownY) or 8
		if rad ~= rad then rad = 8 end
		if rad < 0 then rad = -rad end
		if rad > 20 then rad = 20 end
		if not (hrp.Position.Y >= (TP_FLOOR + rad)) then return end
	end
	local yaw = select(2, hrp.CFrame:ToEulerAnglesYXZ())
	local keep = Vector3.zero
	pcall(function()
		local v = hrp.AssemblyLinearVelocity
		keep = Vector3.new(v.X, 0, v.Z)
	end)
	hrp.CFrame = CFrame.new(hrp.Position.X, TP_FLOOR, hrp.Position.Z) * CFrame.Angles(0, yaw, 0)
	if not force then lastAutoTpAt = tick() end
	if carrying or State.carryMode then
		pcall(function()
			hum:ChangeState(Enum.HumanoidStateType.Running)
			hum.Sit = false
			hrp.AssemblyLinearVelocity = keep
		end)
	else
		pcall(function()
			hrp.Velocity = Vector3.zero
			hrp.AssemblyAngularVelocity = Vector3.zero
			hrp.AssemblyLinearVelocity = Vector3.zero
		end)
	end
end

local AP_L1, AP_L2 = Vector3.new(-476.47, -6.28, 92.73), Vector3.new(-483.12, -4.95, 94.81)

local function startPath(which)
	clearConn("AutoLeft")
	clearConn("AutoRight")
	State.autoLeft = (which == "left")
	State.autoRight = (which == "right")
	local p1 = (which == "left") and AP_L1 or AP_R1
	local p2 = (which == "left") and AP_L2 or AP_R2
	local phase = 1
	local key = (which == "left") and "AutoLeft" or "AutoRight"
	Connections[key] = RunService.Heartbeat:Connect(function()
		if which == "left" and not State.autoLeft then return end
		if which == "right" and not State.autoRight then return end
		if holdingBrainrot() then
			if which == "left" then stopAutoLeft() else stopAutoRight() end
			return
		end
		local hrp, hum = getHRP(), getHum()
		if not hrp or not hum then return end
		if isRagdoll(hum) then hum:Move(Vector3.zero, false) return end
		local tgt = (phase == 1) and p1 or p2
		local flat = Vector3.new(tgt.X - hrp.Position.X, 0, tgt.Z - hrp.Position.Z)
		if flat.Magnitude < 1.15 then
			if phase == 1 then
				phase = 2
			else
				hum:Move(Vector3.zero, false)
				hrp.AssemblyLinearVelocity = Vector3.zero
				if which == "left" then stopAutoLeft() else stopAutoRight() end
			end
			return
		end
		local mv = flat.Unit
		hum:Move(mv, false)
		local spd = pathMoveSpeed()
		pcall(function()
			local v = hrp.AssemblyLinearVelocity
			local hy = (v.Y == v.Y) and v.Y or 0
			hrp.AssemblyLinearVelocity = Vector3.new(mv.X * spd, hy, mv.Z * spd)
		end)
	end)
end

local hittingCooldown = false
local function getClosestTarget()
	local root = getHRP()
	if not root then return nil, math.huge end
	local closest, minDist = nil, math.huge
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP and plr.Character then
			local tRoot = plr.Character:FindFirstChild("HumanoidRootPart")
			local hum = plr.Character:FindFirstChildOfClass("Humanoid")
			if tRoot and hum and hum.Health > 0 then
				local dist = (tRoot.Position - root.Position).Magnitude
				if dist < minDist then minDist = dist; closest = tRoot end
			end
		end
	end
	return closest, minDist
end

local function trySwing()
	if hittingCooldown then return end
	hittingCooldown = true
	local char = LP.Character
	if char then
		local bat = findBat()
		if bat then
			if bat.Parent ~= char then
				local hum = getHum()
				if hum then pcall(function() hum:EquipTool(bat) end) end
			end
			swingBat(bat)
		end
	end
	task.delay(0.35, function() hittingCooldown = false end)
end

local function startAutoBat()
	clearConn("AutoBat")
	local hum0 = getHum()
	if hum0 then hum0.AutoRotate = false end
	Connections.AutoBat = RunService.RenderStepped:Connect(function()
		if not State.autoBat or holdingBrainrot() then return end
		local char = LP.Character
		local root = char and char:FindFirstChild("HumanoidRootPart")
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if not root or not hum then return end
		if not char:FindFirstChildOfClass("Tool") then
			local bat = findBat()
			if bat then pcall(function() hum:EquipTool(bat) end) end
		end
		local target, dist = getClosestTarget()
		if not target then return end
		local myPos, targetPos = root.Position, target.Position
		local targetVel = target.AssemblyLinearVelocity
		local predictPos = targetPos
		if State.predict then
			predictPos = targetPos + targetVel * 0.14 + target.CFrame.LookVector * 0.3
		end
		local direction = predictPos - myPos
		local flatDir = Vector3.new(direction.X, 0, direction.Z)
		if flatDir.Magnitude > 0 then flatDir = flatDir.Unit else flatDir = Vector3.zero end
		local chase = State.speedEnabled and State.customSpeed or 58
		local yVel = ((targetPos.Y + 3.7) - myPos.Y) * 19.5
		if State.predict then yVel += targetVel.Y * 0.8 end
		if hum.FloorMaterial ~= Enum.Material.Air then yVel = math.max(yVel, 13) end
		yVel = math.clamp(yVel, -70, 110)
		root.AssemblyLinearVelocity = root.AssemblyLinearVelocity:Lerp(Vector3.new(flatDir.X * chase, yVel, flatDir.Z * chase), 0.8)
		local lookAt = State.predict and (targetPos + targetVel * math.clamp(targetVel.Magnitude / 150, 0.05, 0.2)) or targetPos
		if (lookAt - myPos).Magnitude > 0.1 then
			local goalCF = CFrame.lookAt(myPos, lookAt)
			local diffCF = root.CFrame:Inverse() * goalCF
			local rx, ry, rz = diffCF:ToEulerAnglesXYZ()
			root.AssemblyAngularVelocity = root.CFrame:VectorToWorldSpace(Vector3.new(
				math.clamp(rx, -2.5, 2.5) * 42,
				math.clamp(ry, -2.5, 2.5) * 42,
				math.clamp(rz, -2.5, 2.5) * 42
		end
		if dist <= 8 then trySwing() end
	end)
end

local stealLast = 0
local stealBusy = false
local stealIdleAnchor = nil
local STEAL_RADIUS = 12
local STEAL_PRIME = 80
local STEAL_HOLD_MIN = 1.3
local STEAL_HOLD_MAX = 2.6
local STEAL_ENTRY_DELAY = 0.3

local function promptPos(prompt)
	if not prompt or not prompt.Parent then return nil end
	if prompt.Parent:IsA("Attachment") then return prompt.Parent.WorldPosition end
	if prompt.Parent:IsA("BasePart") then return prompt.Parent.Position end
	local ok, cf = pcall(function() return prompt.Parent:GetPivot() end)
	if ok and cf then return cf.Position end
	local part = prompt.Parent:FindFirstChildWhichIsA("BasePart")
	return part and part.Position or nil
end

local function stealDist(prompt)
	local root, pos = getHRP(), promptPos(prompt)
	if not root or not pos then return math.huge end
	return (root.Position - pos).Magnitude
end

local function plotIsMine(plot)
	local sign = plot:FindFirstChild("PlotSign")
	local base = sign and sign:FindFirstChild("YourBase")
	return base and base:IsA("BillboardGui") and base.Enabled
end

local function findStealPrompt(maxD)
	local root = getHRP()
	if not root then return nil, math.huge end
	local plots = workspace:FindFirstChild("Plots")
	if not plots then return nil, math.huge end
	local best, bestD = nil, maxD or STEAL_PRIME
	for _, plot in ipairs(plots:GetChildren()) do
		if not plotIsMine(plot) then
			local podiums = plot:FindFirstChild("AnimalPodiums")
			if podiums then
				for _, pod in ipairs(podiums:GetChildren()) do
					for _, d in ipairs(pod:GetDescendants()) do
						if d:IsA("ProximityPrompt") then
							local pos = promptPos(d)
							if pos then
								local dist = (root.Position - pos).Magnitude
								if dist < bestD then best, bestD = d, dist end
							end
						end
					end
				end
			end
		end
	end
	return best, bestD
end

local function grabFns(signal)
	local out = {}
	if not signal then return out end
	pcall(function()
		local list
		if getconnections then list = getconnections(signal)
		elseif get_signal_cons then list = get_signal_cons(signal) end
		if type(list) == "table" then
			for _, c in ipairs(list) do
				local fn = c.Function or c.func or c.Fn
				if type(fn) == "function" then table.insert(out, fn) end
			end
		end
	end)
	return out
end

local function fireStealHold(prompt)
	if not prompt then return end
	local fns = grabFns(prompt.PromptButtonHoldBegan)
	if #fns > 0 then
		for _, fn in ipairs(fns) do pcall(fn) end
	else
		pcall(function()
			if prompt.InputHoldBegin then prompt:InputHoldBegin() end
		end)
	end
end

local function fireStealTrigger(prompt)
	if not prompt then return false end
	local fns = grabFns(prompt.Triggered)
	if #fns > 0 then
		for _, fn in ipairs(fns) do pcall(fn) end
		return true
	end
	if fireproximityprompt then
		local ok = pcall(function() fireproximityprompt(prompt) end)
		if ok then return true end
	end
	return pcall(function()
		if prompt.InputHoldEnd then prompt:InputHoldEnd() end
	end)
end

local function startStealIdle()
	clearConn("AutoStealIdle")
	stealIdleAnchor = tick()
	Connections.AutoStealIdle = RunService.Heartbeat:Connect(function()
		if not State.autoSteal then return end
		if stealBusy then return end
		if holdingBrainrot() then
			setStealProgress(0, "CARRYING")
			stealIdleAnchor = nil
			return
		end
		local now = tick()
		if not stealIdleAnchor then stealIdleAnchor = now end
		setStealProgress(((now - stealIdleAnchor) / STEAL_HOLD_MAX) % 1, "SEARCHING")
	end)
end

local function startAutoSteal()
	clearConn("AutoSteal")
	stealBusy = false
	setStealProgress(0, "SEARCHING")
	startStealIdle()
	Connections.AutoSteal = RunService.Heartbeat:Connect(function()
		if not State.autoSteal then return end
		if stealBusy then return end
		if holdingBrainrot() then return end
		if tick() - stealLast < 0.08 then return end
		local prompt = select(1, findStealPrompt(STEAL_PRIME))
		if not prompt then return end
		stealBusy = true
		task.spawn(function()
			pcall(function()
				local current = prompt
				fireStealHold(current)
				local t0 = tick()
				while State.autoSteal and current.Parent and tick() - t0 < STEAL_HOLD_MIN do
					setStealProgress((tick() - t0) / STEAL_HOLD_MAX, "HOLDING")
					task.wait(0.02)
				end
				local alreadyIn = stealDist(current) <= STEAL_RADIUS
				local fired = false
				while State.autoSteal and current.Parent and tick() - t0 <= STEAL_HOLD_MAX do
					local elapsed = tick() - t0
					local live, liveD = findStealPrompt(STEAL_RADIUS)
					if live and live.Parent and liveD <= STEAL_RADIUS then
						setStealProgress(math.clamp(elapsed / STEAL_HOLD_MAX, 0, 1), "STEALING")
						if live ~= current then
							current = live
							fireStealHold(current)
							task.wait(0.1)
						elseif not alreadyIn then
							task.wait(STEAL_ENTRY_DELAY)
						end
						if State.autoSteal and current and current.Parent and stealDist(current) <= STEAL_RADIUS then
							fireStealTrigger(current)
							fired = true
							stealLast = tick()
							setStealProgress(1, "STOLE")
						end
						break
					end
					setStealProgress(math.clamp(elapsed / STEAL_HOLD_MAX, 0, 1), "HOLDING")
					task.wait(0.02)
				end
				if fired then task.wait(0.4) end
			end)
			stealBusy = false
			stealIdleAnchor = tick()
			if not State.autoSteal then
				setStealProgress(0, "OFF")
			end
		end)
	end)
end
local function stopAutoSteal()
	clearConn("AutoSteal")
	clearConn("AutoStealIdle")
	stealBusy = false
	stealIdleAnchor = nil
	setStealProgress(0, "OFF")
end

local fpsActive = false
local function stripFX(obj)
	pcall(function()
		if obj:IsA("BasePart") then
			obj.CastShadow = false
			if obj.Material ~= Enum.Material.SmoothPlastic and obj.Material ~= Enum.Material.Neon then
				obj.Material = Enum.Material.SmoothPlastic
			end
		elseif obj:IsA("Decal") or obj:IsA("Texture") then
			obj.Transparency = 1
		elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") or obj:IsA("Smoke") or obj:IsA("Fire") then
			obj.Enabled = false
		elseif obj:IsA("PostEffect") then
			obj.Enabled = false
		end
	end)
end

local function createKeybind(name, label, stateKey, parent)
	local btn = createPill(name, label .. ": ?", parent)
	local function refresh()
		local k = State[stateKey]
		local n = (typeof(k) == "EnumItem" and k ~= Enum.KeyCode.Unknown) and k.Name or "?"
		btn.Text = label .. ": " .. n
	end
	btn:SetAttribute("RefreshKey", true)
	refresh()
	btn.MouseButton1Click:Connect(function()
		btn.Text = label .. ": ..."
		local conn
		conn = UserInputService.InputBegan:Connect(function(inp, gp)
			if inp.UserInputType ~= Enum.UserInputType.Keyboard then return end
			if conn then conn:Disconnect() end
			if inp.KeyCode == Enum.KeyCode.Escape then refresh() return end
			State[stateKey] = inp.KeyCode
			refresh()
		end)
	end)
	return btn, refresh
end

local function makeDraggable(frame, handle, respectLock, saveKey)
	local dragging, dragStart, startPos = false, nil, nil
	handle.InputBegan:Connect(function(input)
		if dragging or (respectLock and State.guiLocked) then return end
		if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
		dragging = true
		dragStart = input.Position
		startPos = frame.Position
	end)
	UserInputService.InputChanged:Connect(function(input)
		if not dragging then return end
		if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then return end
		local delta = input.Position - dragStart
		frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			if dragging then
				dragging = false
				if saveKey then saveUDim(saveKey, frame.Position) end
			end
		end
	end)
end

local OB_W, OB_H, GAP = 84, 26, 5
local function makeOverlay(key, text, col, row, onClick)
	local x = -(OB_W + 10) - (col - 1) * (OB_W + GAP)
	local yOff = (row - 3) * (OB_H + GAP)
	local container = Instance.new("Frame")
	container.Name = "Btn_" .. tostring(key)
	container.Parent = overlayGui
	container.BackgroundColor3 = BLACK
	container.BorderSizePixel = 0
	container.Size = UDim2.new(0, OB_W, 0, OB_H)
	container.Position = loadUDim("RevulBtn_" .. tostring(key) .. ".txt", UDim2.new(1, x, 0.5, yOff))
	container.Active = true
	pillCorner(container)
	addStroke(container, 1.5)
	local btnScale = Instance.new("UIScale")
	btnScale.Name = "BtnScale"
	btnScale.Scale = tonumber(State.btnScale) or 1
	btnScale.Parent = container

	local lbl = Instance.new("TextLabel")
	lbl.Size = UDim2.new(1, -6, 1, 0)
	lbl.Position = UDim2.new(0, 3, 0, 0)
	lbl.BackgroundTransparency = 1
	lbl.Font = Enum.Font.GothamBold
	lbl.Text = text
	lbl.TextColor3 = WHITE
	lbl.TextSize = 9
	lbl.TextWrapped = true
	lbl.ZIndex = 3
	lbl.Parent = container

	local click = Instance.new("TextButton")
	click.Size = UDim2.new(1, 0, 1, 0)
	click.BackgroundTransparency = 1
	click.Text = ""
	click.ZIndex = 5
	click.AutoButtonColor = false
	click.Parent = container

	local active = false
	local function setActive(state)
		active = state == true
		container.BackgroundColor3 = active and BLACK_ON or BLACK
		local st = container:FindFirstChildOfClass("UIStroke")
		if st then st.Transparency = active and 0 or 0.35 end
	end

	local dragging, moved, pressPos, origin = false, false, nil, nil
	click.InputBegan:Connect(function(input)
		if State.overlayLocked then return end
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			moved = false
			pressPos = input.Position
			origin = container.Position
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if not dragging then return end
		if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then return end
		local delta = input.Position - pressPos
		if not moved then
			if math.abs(delta.X) < 6 and math.abs(delta.Y) < 6 then return end
			moved = true
		end
		container.Position = UDim2.new(origin.X.Scale, origin.X.Offset + delta.X, origin.Y.Scale, origin.Y.Offset + delta.Y)
	end)
	local function finishDrag()
		if not dragging then return end
		dragging = false
		if moved then
			saveUDim("RevulBtn_" .. tostring(key) .. ".txt", container.Position)
		end
	end
	click.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			finishDrag()
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			finishDrag()
		end
	end)
	click.MouseButton1Click:Connect(function()
		if moved then moved = false return end
		if isSafeLocked() then return end
		if onClick then onClick(setActive, active) end
	end)

	Overlay.Frames[key] = container
	Overlay.Buttons[key] = setActive
	return setActive
end

local function refreshOverlayVis()
	local map = {
		drop = State.showDrop,
		autoLeft = State.showAutoLeft,
		autoRight = State.showAutoRight,
		aimbot = State.showAimbot,
		tpDown = State.showTPDown,
		carry = State.showCarry,
		lagger = State.showLagger,
		reset = State.showReset,
	for key, vis in pairs(map) do
		local f = Overlay.Frames[key]
		if f then f.Visible = vis end
	end
end

local function flash(setA)
	setA(true)
	task.delay(0.28, function() setA(false) end)
end

local function setBoth(pill, overlayKey, on)
	setPillOn(pill, on)
	if Overlay.Buttons[overlayKey] then Overlay.Buttons[overlayKey](on) end
end

syncVisuals = function(key)
	if key == "autoLeft" then setPillOn(P.autoLeft, State.autoLeft) end
	if key == "autoRight" then setPillOn(P.autoRight, State.autoRight) end
end

local function toggleFlag(flag, pill, overlayKey, onEnable, onDisable)
	State[flag] = not State[flag]
	local on = State[flag]
	setBoth(pill, overlayKey, on)
	if on then
		if onEnable then onEnable() end
	else
		if onDisable then onDisable() end
	end
end

local function toggleCarry()
	toggleFlag("carryMode", P.carry, "carry")
	if State.carryMode then
		State.lagger = false
		setBoth(P.lagger, "lagger", false)
	elseif not State.speedEnabled then
		restoreWalkSpeed()
	end
end

local function toggleLagger()
	toggleFlag("lagger", P.lagger, "lagger")
	if State.lagger then
		State.carryMode = false
		setBoth(P.carry, "carry", false)
	elseif not State.speedEnabled then
		restoreWalkSpeed()
	end
end

local function wireShow(pill, flag, overlayKey)

		local on = State[flag] == true
		setPillOn(pill, on)

		end
	end
	apply()
	pill.MouseButton1Click:Connect(function()
		State[flag] = not State[flag]
		apply()
	end)
end

RunService.Heartbeat:Connect(function()
	local hum, root = getHum(), getHRP()
	if not root then return end
	local pathing = State.autoLeft or State.autoRight
	local moving = pathing or (hum and hum.MoveDirection and hum.MoveDirection.Magnitude > 0.05)
	local shown = 0
	if moving then
		if pathing then
			shown = pathMoveSpeed()
		elseif State.lagger then
			shown = tonumber(State.laggerSpeed) or 15
		elseif State.carryMode or (State.autoCarrySwitch and holdingBrainrot()) then
			shown = tonumber(State.carrySpeed) or 30
		elseif State.speedEnabled then
			shown = tonumber(State.customSpeed) or 29
		else
			shown = tonumber(hum.WalkSpeed) or DEFAULT_WALK
		end
	end
	if shown ~= shown then shown = 0 end
	local txt = "SPEED: " .. string.format("%.1f", shown)
	SpeedLabel.Text = txt
	if SpeedHud.label and SpeedHud.label.Parent then
		SpeedHud.label.Text = "SPEED " .. string.format("%.1f", shown)
	end
end)

local function fireOverlayKey(which)
	if isSafeLocked() then return end
	if which == "drop" then dropBrainrot()
	elseif which == "autoLeft" then
		if State.autoLeft then stopAutoLeft(); setPillOn(P.autoLeft, false)
		else startPath("left"); setBoth(P.autoLeft, "autoLeft", true); setBoth(P.autoRight, "autoRight", false) end
	elseif which == "autoRight" then
		if State.autoRight then stopAutoRight(); setPillOn(P.autoRight, false)
		else startPath("right"); setBoth(P.autoRight, "autoRight", true); setBoth(P.autoLeft, "autoLeft", false) end
	elseif which == "aimbot" then
		toggleFlag("autoBat", P.autoBat, "aimbot", function()
			if State.autoLeft then stopAutoLeft() end
			if State.autoRight then stopAutoRight() end
			startAutoBat()
		end, stopAutoBat)
	elseif which == "tpDown" then tpDownNow(true)
	elseif which == "carry" then toggleCarry()
	elseif which == "lagger" then toggleLagger()
	elseif which == "reset" then doInstantReset()
	end
end

local function loadConfig()
	local raw = fileRead(CONFIG_FILE)
	if type(raw) ~= "string" or raw == "" then return end

	if not ok or type(data) ~= "table" then return end
	local function num(k, lo, hi)
		if data[k] ~= nil then
			local n = tonumber(data[k])
			if n then State[k] = math.clamp(n, lo, hi) end
		end
	end
	num("customSpeed", 1, 500); num("carrySpeed", 1, 500); num("laggerSpeed", 1, 200)
	num("fov", 30, 120); num("autoTpDownY", 0, 40); num("guiScale", 0.45, 1.4); num("btnScale", 0.4, 2.5)
	for _, k in ipairs({"showDrop","showAutoLeft","showAutoRight","showAimbot","showTPDown","showCarry","showLagger","showReset","showStealBar","showSpeedCounter","overlayLocked","speedEnabled","carryMode","lagger","infJump","unwalk","noCollision","antiRagdoll","antiDie","autoBat","autoSteal","predict","batCounter","fpsBoost","shaders","lowGfx","optimizer","galaxySky","autoTpDown","autoCarrySwitch","safeMode"}) do
		if data[k] ~= nil then State[k] = data[k] and true or false end
	end
	if type(data.keys) == "table" then

			pcall(function()

	local function getCountdownLabel()
		local ok, label = pcall(function()
			local pg = LP:FindFirstChild("PlayerGui")
			if not pg then return nil end
			local top = pg:FindFirstChild("DuelsMachineTopFrame", true)
			if not top then return nil end
			local timer = top:FindFirstChild("Timer", true)
			if not timer then return nil end
			return timer:FindFirstChild("Label", true) or timer:FindFirstChildWhichIsA("TextLabel", true)
		end)
		return (ok and label) or nil
	end
	local function countdownValue(text)
		local value = tostring(text or ""):upper():gsub("^%s+", ""):gsub("%s+$", "")
		if value == "GO" or value == "GO!" or value == "START" or value == "START!" then return nil end
		if value:find(":", 1, true) then return nil end
		local number = tonumber(value)
		if not number then
			for token in value:gmatch("%d+%.?%d*") do number = tonumber(token) end
		end
		if number ~= nil and number >= 0 and number <= 5 then return number end
		return nil
	end
